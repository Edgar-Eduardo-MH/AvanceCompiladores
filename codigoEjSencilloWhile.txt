EJEMPLO WHILE

int main() {
    int a = 0;
    cout << "Escribe un numero: ";
    cin >> a;
    
    if (a > 10) then
        cout << "Tu numero es mayor que 10.";
    else
        cout << "Tu numero es 10 o menos.";
    end
    
    int i = 0;
    while (i < a) then
        cout << "je";
        i = i + 1;
    end
    
    cout << "Conteo finalizado.";
}

EJEMPLO SWITCH

int main() {
    int x = 1;
    
    cout << "Iniciando Do-Until...";
    
    do
        cout << x;
        switch (x)
            case 1 then
                cout << " (uno)";
            case 2 then
                cout << " (dos)";
            case 3 then
                cout << " (tres)";
        end
        x = x + 1;
    until (x > 3);
    
    cout << "Fin.";
}

EJEMPLO FOR

int main() {
    cout << "Iniciando bucle for:";
    
    for (int i = 0; i < 5; i = i + 1;) then
        cout << "Iteracion: ";
        cout << i;
    end
    
    cout << "Bucle for finalizado.";
}
